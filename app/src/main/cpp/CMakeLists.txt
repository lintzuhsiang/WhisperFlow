cmake_minimum_required(VERSION 3.22.1)
project("whisper_jni" C CXX)

# ── Core ggml + Whisper sources ────────────────────────────────────────────
set(WHISPER_JNI_SOURCES
    # JNI bridge
    whisper_jni.cpp

    # Whisper core
    whisper.cpp

    # GGML tensor library
    ggml.c
    ggml-alloc.c
    ggml-backend.cpp          # NOTE: .cpp not .c
    ggml-backend-reg.cpp
    ggml-backend-dl.cpp       # dl_load_library / dl_get_sym / dl_error (POSIX dlfcn)
    ggml-quants.c
    ggml-threading.cpp

    # CPU backend – implementation (ggml-cpu/ subdirectory)
    ggml-cpu/ggml-cpu.c
    ggml-cpu/ggml-cpu.cpp
    ggml-cpu/repack.cpp
    ggml-cpu/hbm.cpp
    ggml-cpu/quants.c
    ggml-cpu/traits.cpp
    ggml-cpu/binary-ops.cpp
    ggml-cpu/unary-ops.cpp
    ggml-cpu/vec.cpp
    ggml-cpu/ops.cpp
    ggml-cpu/llamafile/sgemm.cpp
)

# ── Architecture-specific sources ─────────────────────────────────────────
# Android phones are ARM64; the emulator is x86_64.
if (ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "armeabi-v7a")
    list(APPEND WHISPER_JNI_SOURCES
        ggml-cpu/arch/arm/quants.c
        ggml-cpu/arch/arm/repack.cpp
    )
elseif (ANDROID_ABI STREQUAL "x86_64" OR ANDROID_ABI STREQUAL "x86")
    list(APPEND WHISPER_JNI_SOURCES
        ggml-cpu/arch/x86/quants.c
        ggml-cpu/arch/x86/repack.cpp
    )
endif()

add_library(whisper_jni SHARED ${WHISPER_JNI_SOURCES})

# ── Language standards ─────────────────────────────────────────────────────
target_compile_features(whisper_jni PRIVATE c_std_11 cxx_std_17)

# ── Compile definitions ────────────────────────────────────────────────────
# GGML_VERSION / GGML_COMMIT are normally generated by CMake via configure_file
# from the ggml repo's ggml-version.h.in.  Since we vendor the source directly
# we provide fixed placeholder values here.
# GGML_USE_CPU must be defined so ggml-backend-reg.cpp includes ggml-cpu.h and
# calls ggml_backend_cpu_reg() – without it the CPU backend is never registered
# and all inference silently produces no output.
target_compile_definitions(whisper_jni PRIVATE
    GGML_VERSION="0.0.0"
    GGML_COMMIT="unknown"
    GGML_USE_CPU
)

# ── Base compile flags ─────────────────────────────────────────────────────
target_compile_options(whisper_jni PRIVATE
    -O3
    -DNDEBUG
    -DGGML_USE_LLAMAFILE        # enable llamafile SGEMM (already in sources)
)

# ── ABI-specific SIMD flags ────────────────────────────────────────────────
if (ANDROID_ABI STREQUAL "arm64-v8a")
    # armv8-a+dotprod+fp16 gives good acceleration for Q8_0/F16 whisper models.
    # NOTE: +i8mm (armv8.2-a Int8 matrix multiply) was removed because it introduces
    # strict memory-alignment requirements that the flash-attention tiled wdata scratch
    # layout does not guarantee in multi-threaded inference, causing SIGSEGV in
    # ggml_compute_forward_flash_attn_ext on Qualcomm Adreno (Snapdragon 8 Gen series).
    target_compile_options(whisper_jni PRIVATE -march=armv8-a+dotprod+fp16)
elseif (ANDROID_ABI STREQUAL "x86_64")
    target_compile_options(whisper_jni PRIVATE -msse4.2 -mavx -mavx2 -mfma -mf16c)
endif()

# ── Include directories ────────────────────────────────────────────────────
target_include_directories(whisper_jni PRIVATE
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/ggml-cpu
)

# ── Android system libraries ───────────────────────────────────────────────
find_library(log-lib  log)
find_library(android-lib android)
target_link_libraries(whisper_jni ${log-lib} ${android-lib})


# ══════════════════════════════════════════════════════════════════════════════
#  SentencePiece – NLLB tokenizer (JNI bridge)
# ══════════════════════════════════════════════════════════════════════════════
#
# Fetches Google SentencePiece at configure time and builds a static library.
# We then wrap it in a tiny SHARED JNI library (sentencepiece_jni) that
# Kotlin's SentencePieceJni class loads via System.loadLibrary().
# ──────────────────────────────────────────────────────────────────────────────

include(FetchContent)
FetchContent_Declare(
    sentencepiece
    GIT_REPOSITORY https://github.com/google/sentencepiece.git
    GIT_TAG        v0.2.0
    GIT_SHALLOW    TRUE
)

# Disable everything we don't need – keeps the build fast and small.
set(SPM_ENABLE_SHARED  OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST     OFF CACHE BOOL "" FORCE)
set(SPM_COVERAGE       OFF CACHE BOOL "" FORCE)

# CMake 4.x removed compat for cmake_minimum_required(VERSION <3.5).
# This variable (CMake 3.31+) clamps subproject minimums so configure succeeds.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)

# ── Patch-then-populate approach ──────────────────────────────────────────────
#
# Problem: SentencePiece's CMakeLists.txt contains:
#     add_compile_options(-fmacro-prefix-map=${CMAKE_SOURCE_DIR}/='')
#
# Inside FetchContent, CMAKE_SOURCE_DIR resolves to OUR top-level source dir
# (".../Mobile Live Tranlation/app/src/main/cpp").  The space in that path
# causes clang to split the argument and fail with:
#     error: no such file or directory: 'Tranlation/app/src/main/cpp/='
#
# Fix: download the source first (FetchContent_Populate), strip that one line
# with a pure-CMake file() regex, then hand off to add_subdirectory().
# ──────────────────────────────────────────────────────────────────────────────
FetchContent_GetProperties(sentencepiece)
if(NOT sentencepiece_POPULATED)
    FetchContent_Populate(sentencepiece)

    # Remove any -fmacro-prefix-map lines before CMake processes the subproject.
    file(READ "${sentencepiece_SOURCE_DIR}/CMakeLists.txt" _spm_txt)
    string(REGEX REPLACE
        "[^\n]*-fmacro-prefix-map[^\n]*\n?"
        ""
        _spm_txt "${_spm_txt}")
    file(WRITE "${sentencepiece_SOURCE_DIR}/CMakeLists.txt" "${_spm_txt}")

    # Fix: SentencePiece builds internal tools (spm_decode) that use protobuf,
    # which uses Android logging. Link liblog globally so they build successfully.
    link_libraries(log)

    add_subdirectory(
        "${sentencepiece_SOURCE_DIR}"
        "${sentencepiece_BINARY_DIR}"
        EXCLUDE_FROM_ALL
    )
endif()

add_library(sentencepiece_jni SHARED sentencepiece_jni.cpp)
target_compile_features(sentencepiece_jni PRIVATE cxx_std_17)
target_compile_options(sentencepiece_jni PRIVATE -O2 -DNDEBUG)

# Tell our JNI wrapper where the SentencePiece C++ headers are
target_include_directories(sentencepiece_jni PRIVATE "${sentencepiece_SOURCE_DIR}/src")

# sentencepiece-static exports its include dirs, but sometimes CMake interface
# propagation fails across FetchContent/add_subdirectory bounds.
target_link_libraries(sentencepiece_jni
    sentencepiece-static
    ${log-lib}
)
